    <script>
        (function () {
            function switchLang(lang) {
                var url = new URL(window.location.href);
                url.searchParams.set('lang', lang);
                window.location.href = url.toString();
            }

            // Theme-aware favicon: the TAB BAR follows the browser/OS scheme (not the site's
            // pcp-theme toggle) — dark strokes for light tab bars, light strokes for dark ones.
            // /favicon.ico stays as the light-tab fallback for browsers that never run this.
            function applyFavicon() {
                var dark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
                var href = dark ? '/img/favicon-tab-dark.png' : '/img/favicon-tab-light.png';
                var link = document.querySelector('link[rel="icon"]');
                if (!link) {
                    link = document.createElement('link');
                    link.rel = 'icon';
                    link.type = 'image/png';
                    document.head.appendChild(link);
                }
                if (link.getAttribute('href') !== href) {
                    link.setAttribute('href', href);
                }
            }
            applyFavicon();
            if (window.matchMedia) {
                try {
                    window.matchMedia('(prefers-color-scheme: dark)')
                        .addEventListener('change', applyFavicon);
                } catch (e) { /* older Safari: no change events — startup pick stands */ }
            }

            document.addEventListener('DOMContentLoaded', function () {
                var links = document.querySelectorAll('#lang-switcher [data-lang]');
                links.forEach(function (link) {
                    link.addEventListener('click', function (e) {
                        e.preventDefault();
                        var lang = link.getAttribute('data-lang');
                        switchLang(lang);
                    });
                });

                // MA-1: sticky bars (lesson editor Save bar, outline rail) sit under the navbar's
                // real height, which differs between the collapsed phone bar and desktop.
                var nav = document.getElementById('main-navbar');
                if (nav) {
                    var setNavHeight = function () {
                        document.documentElement.style.setProperty('--pcp-nav-h', nav.offsetHeight + 'px');
                    };
                    setNavHeight();
                    if (window.ResizeObserver) {
                        new ResizeObserver(setNavHeight).observe(nav);
                    } else {
                        window.addEventListener('resize', setNavHeight);
                    }
                }

                // MA-1: pages that read like an app on a phone (lesson player) ask for a navbar that
                // slides away while scrolling down and comes back on scroll up.
                if (nav && document.body.hasAttribute('data-nav-autohide') && window.matchMedia
                        && window.matchMedia('(max-width: 767.98px)').matches) {
                    nav.classList.add('navbar-autohide');
                    var menu = document.getElementById('navbarLanding');
                    var lastY = window.scrollY;
                    window.addEventListener('scroll', function () {
                        var y = window.scrollY;
                        var menuOpen = menu && (menu.classList.contains('show') || menu.classList.contains('collapsing'));
                        if (menuOpen || y < nav.offsetHeight) {
                            nav.classList.remove('nav-hidden');
                        } else if (y > lastY + 6) {
                            nav.classList.add('nav-hidden');
                        } else if (y < lastY - 6) {
                            nav.classList.remove('nav-hidden');
                        }
                        lastY = y;
                    }, { passive: true });
                }

                // QW-1: theme toggle — flip data-bs-theme, persist, and notify (Ace listens).
                var toggle = document.getElementById('theme-toggle');
                if (toggle) {
                    toggle.addEventListener('click', function () {
                        var cur = document.documentElement.getAttribute('data-bs-theme') === 'dark'
                            ? 'dark' : 'light';
                        var next = cur === 'dark' ? 'light' : 'dark';
                        document.documentElement.setAttribute('data-bs-theme', next);
                        try { localStorage.setItem('pcp-theme', next); } catch (e) { /* ignore */ }
                        document.dispatchEvent(new CustomEvent('pcp:theme', { detail: { theme: next } }));
                    });
                }
            });
        })();
    </script>
